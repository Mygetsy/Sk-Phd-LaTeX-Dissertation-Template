\begin{document}
%%% Переопределение именований типовых разделов
% https://tex.stackexchange.com/a/156050
% Имена привязываем к главному языку работы (русский или английский),
% содержимое renames.tex само выбирает язык по счётчику thesislang.
\ifnumequal{\value{thesislang}}{1}{%
  \gappto\captionsenglish{\input{common/renames}}% for polyglossia and babel
  \gappto\captionsenglish{\input{Dissertation/renames}}%
}{%
  \gappto\captionsrussian{\input{common/renames}}% for polyglossia and babel
  \gappto\captionsrussian{\input{Dissertation/renames}}%
}
\input{common/renames}
\input{Dissertation/renames}

%%% Структура диссертации (ГОСТ Р 7.0.11-2011, 4)
% Обязательный русский титульный лист --- в обеих версиях (русской и английской).
% ВАЖНО: \include нельзя помещать в аргумент \ifnumequal, поэтому используем
% примитивный \ifnum...\fi (иначе \include ломает вёрстку и переключение языка).
\ifnum\value{thesislang}=1 \selectlanguage{russian}\fi
\include{Dissertation/title_rus}       % 1. Русский титульный лист
\ifnum\value{thesislang}=1
  \selectlanguage{english}%
  \include{Dissertation/title}         % 2. Английский титульный лист
  \selectlanguage{russian}%
  \include{Dissertation/abstract_rus}  % 3. Русская аннотация
  \selectlanguage{english}%
  \include{Dissertation/abstract}      % 4. Английская аннотация
\fi
\include{Dissertation/contents}        % Оглавление
\ifnumequal{\value{contnumfig}}{1}{}{\counterwithout{figure}{chapter}}
\ifnumequal{\value{contnumtab}}{1}{}{\counterwithout{table}{chapter}}
\include{Dissertation/introduction}    % Введение
\ifnumequal{\value{contnumfig}}{1}{\counterwithout{figure}{chapter}
}{\counterwithin{figure}{chapter}}
\ifnumequal{\value{contnumtab}}{1}{\counterwithout{table}{chapter}
}{\counterwithin{table}{chapter}}
